\entete{M2}{Dix mesures, une cale}{Si je mesure dix fois la même cale, est-ce que je trouve dix fois pareil ?}{Le moyen de mesure}{20 min}{en binôme}{1}
\begin{tcolorbox}[boite]\textbf{Le but.} Découvrir la dispersion d'une mesure et son écart à la vraie valeur, sans aucun logiciel.\par\textbf{Il vous faut.} Un micromètre 0-25 mm · Une cale étalon de 10 mm · Gants ou chiffon\end{tcolorbox}
\begin{tcolorbox}[pred]Sur dix mesures de la même cale au micromètre, combien de valeurs différentes pensez-vous obtenir ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item Nettoyer la cale. La tenir par les bords, sans la garder dans la main.
\item L'un mesure, l'autre note. Mesurer la cale 10 fois, au cliquet. \textbf{Retirer puis remettre} la cale avant chaque mesure.
\item Noter toutes les valeurs, même si l'une paraît bizarre.
\item Calculer la moyenne, l'étendue (plus grande moins plus petite) et l'écart entre la moyenne et la valeur de la cale.
\end{enumerate}

\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item Moyenne = \champ{18mm} \quad Étendue = \champ{18mm} \quad Moyenne $-$ valeur de la cale = \champ{18mm}\par\vspace{5pt}
\item Vos mesures sont-elles groupées ? Sont-elles centrées sur la valeur de la cale ?\reponse[2]
\item Pourquoi retirer la cale entre deux mesures ?\reponse[1]
\end{enumerate}
\begin{tcolorbox}[retenir]Deux qualités différentes : la \textbf{fidélité} (les mesures sont groupées) et la \textbf{justesse} (elles tombent sur la vraie valeur).\end{tcolorbox}
