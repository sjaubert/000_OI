\entete{A6}{Compter les défauts : la carte c}{Le nombre de défauts par pièce est-il stable ?}{Cartes aux attributs}{30 min}{en binôme}{2}
\begin{tcolorbox}[boite]\textbf{Le but.} Relever des défauts par unité, calculer les limites de la carte c et la tracer.\par\textbf{Il vous faut.} La page « Tirer un lot », mode « défauts par unité » · Ellistat\end{tcolorbox}
\begin{tcolorbox}[pred]On relève en moyenne 3 défauts par panneau. Un panneau avec 8 défauts est-il anormal ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item \tagW\ Page « Tirer un lot » : on compte « des défauts par unité », scénario donné par le formateur. Tirer 20 unités, une par une.
\item Noter chaque résultat et placer le point sur la carte.
\item Calculer $\bar c$ = total des défauts / nombre d'unités, puis les limites $\bar c \pm 3\sqrt{\bar c}$.
\item \tagE\ « Copier le tableau », coller dans une grille. \menu{Stats Desc.} > \menu{Cartes de contrôle aux attributs} : « Attributs » = la colonne des défauts ; laisser « Taille de lot » vide (les unités sont identiques) ; carte \textbf{c}. Dans « Phase I », Début 1 et Fin = le nombre d'unités de votre relevé.
\end{enumerate}
