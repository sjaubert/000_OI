\entete{M10}{Fabriquer son étalon}{Comment obtenir un étalon de 12,000 mm avec des cales ?}{Le moyen de mesure}{25 min}{en binôme}{2}
\begin{tcolorbox}[boite]\textbf{Le but.} Composer une valeur par adhérence de deux cales et s'en servir comme étalon au plus près de la cote contrôlée.\par\textbf{Il vous faut.} Un coffret de cales (cales de 10 et de 2 mm) · Un micromètre 0-25 mm · Chiffon non pelucheux\end{tcolorbox}
\begin{tcolorbox}[pred]Deux cales de 10 et de 2 mm collées l'une sur l'autre font-elles exactement 12,000 mm au micromètre ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item Nettoyer les faces des deux cales. Les poser en croix, presser légèrement et faire glisser jusqu'à ce qu'elles collent : c'est l'\textbf{adhérence}.
\item Mesurer la pile 15 fois au micromètre, en la retirant à chaque fois. Ne pas la tenir dans la main.
\item \tagE\ \menu{Gage R\&R}, onglet \menu{Justesse}, « Ajouter une justesse » : Nombre de mesures 15 ; Valeur vraie 12,000 ; LSI 11,85 et LSS 12,00 ; k 3. Saisir les 15 mesures avec trois décimales.
\item Séparer les cales en les faisant glisser, sans tirer. Les nettoyer et les ranger.
\end{enumerate}

\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item Moyenne de la pile = \champ{18mm} mm. Écart à 12,000 = \champ{18mm} mm.\par\vspace{5pt}
\item L'étalon vaut maintenant 12,000 mm, une limite de la tolérance de la butée. Pourquoi est-ce mieux que la cale de 10 mm ?\reponse[2]
\item Comparez Cg et Cgk avec ceux obtenus sur la cale de 10 mm (M3).\reponse[1]
\end{enumerate}
\begin{tcolorbox}[retenir]On vérifie un instrument là où il sert : avec un étalon proche de la cote à contrôler.\end{tcolorbox}
