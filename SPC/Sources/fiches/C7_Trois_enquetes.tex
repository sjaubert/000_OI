\entete{C7}{Trois enquêtes}{La carte sonne : que s'est-il passé, que fait-on ?}{Cartes aux mesures et capabilité}{60 min}{en binôme}{2}
\begin{tcolorbox}[boite]\textbf{Le but.} Mener une enquête sur des données réelles, décider, puis présenter au groupe en 5 minutes.\par\textbf{Il vous faut.} Un jeu par binôme : Pression, Dimension ou Liquide (page « Données à copier »)\end{tcolorbox}
\begin{tcolorbox}[pred]Avant de tracer la carte : dans le tableau de chiffres, quelles valeurs vous semblent anormales ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item \textbf{Pression} : pression d'un réservoir, 5 relevés par jour pendant 24 jours (Répétition = 5). \textbf{Dimension} : pièce usinée, 5 pièces par heure, 27 sous-groupes (Répétition = 5). \textbf{Liquide} : une valeur par lot, 17 lots (Répétition = 1).
\item \tagE\ Coller le jeu dans une nouvelle grille, régler la Répétition, puis \menu{Stats Desc.} > \menu{Détail}, référentiel \menu{Centré Moyenne}.
\item \tagE\ Repérer les points hors limites et les points marqués « Alerte ».
\item Préparer la présentation : la situation, la carte, ce qui sonne, la décision.
\end{enumerate}
\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item Quels points sortent des limites ? Quels points portent une alerte ?\reponse[2]
\item Que s'est-il passé : un événement ponctuel, un changement durable, une dérive ?\reponse[2]
\item Les limites sont calculées sur toutes les données, y compris les périodes perturbées. Est-ce honnête ?\reponse[2]
\item Quelle réaction proposez-vous au poste ?\reponse[2]
\end{enumerate}
\begin{tcolorbox}[retenir]Une carte se lit comme une histoire : quand, quoi, pourquoi. Puis on décide.\end{tcolorbox}
