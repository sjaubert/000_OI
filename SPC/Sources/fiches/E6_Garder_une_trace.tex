\entete{E6}{Garder une trace}{Comment retrouver et montrer un résultat ?}{Prise en main d'Ellistat}{10 min}{seul}{1}
\begin{tcolorbox}[boite]\textbf{Le but.} Enregistrer un graphique, le retrouver et l'exporter.\par\textbf{Il vous faut.} Un graphique déjà tracé (E4 ou E5)\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item \tagE\ Survoler un graphique : une barre d'outils apparaît en haut à droite. Cliquer sur la disquette \menu{Enregistrer dans mes graphiques}, donner un titre et un commentaire.
\item \tagE\ En haut de la page de grille, cliquer sur \menu{Mes graphiques} : la bibliothèque est rangée par jour.
\item \tagE\ Cliquer sur \menu{Exporter en PowerPoint}, sélectionner le graphique, exporter.
\item \tagE\ \menu{Renommer la grille} avec un nom parlant (par exemple « Butées 6 x 5 »).
\end{enumerate}
\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item Vous corrigez une valeur dans la grille après avoir enregistré le graphique. Le graphique enregistré change-t-il ?\reponse[1]
\item Quel titre donner à un graphique pour qu'un collègue le comprenne sans vous ?\reponse[1]
\end{enumerate}
\begin{tcolorbox}[retenir]Un graphique enregistré est une photo : il ne suit pas les modifications de la grille. On le date et on le nomme.\end{tcolorbox}
\begin{tcolorbox}[plus]Dans \menu{Liste de mes projets}, ajouter le formateur aux « utilisateurs autorisés » de votre projet.\end{tcolorbox}
