\stitre{1. Le modèle}
$n$ pièces, $k$ opérateurs, $r$ répétitions (exemple Bosch : $n=10$, $k=3$, $r=2$). La mesure de la pièce $i$ par l'opérateur $j$, à la répétition $l$ :
\begin{center}\vspace{-3pt}$x_{ijl}=\mu+P_i+O_j+(PO)_{ij}+\varepsilon_{ijl}$\vspace{-2pt}\end{center}
Les quatre termes sont aléatoires, de variances $\sigma_P^2$ (pièces), $\sigma_O^2$ (opérateurs), $\sigma_{PO}^2$ (interaction pièce $\times$ opérateur) et $\sigma^2$ (répétition). Ces variances s'additionnent, pas les écarts types.\par\vspace{3pt}
\renewcommand{\arraystretch}{1.08}
\begin{tabularx}{\linewidth}{|>{\bfseries}p{37mm}|>{\raggedright\arraybackslash}X|}\hline
\rowcolor{uimm}\color{white}\textbf{Nom} & \color{white}\textbf{Définition}\\\hline
EV, répétabilité & $\sigma$, l'écart type des mesures répétées (même pièce, même opérateur)\\\hline
AV, reproductibilité & $\sigma_O$, l'écart type entre opérateurs (avec $\sigma_{PO}$ si l'interaction n'est pas regroupée)\\\hline
PV, pièces & $\sigma_P$, l'écart type entre pièces\\\hline
GRR & $\sqrt{\mathrm{EV}^2+\mathrm{AV}^2}$ \quad (le «\,Gage R\&R\,», en unité de mesure)\\\hline
TV, total & $\sqrt{\mathrm{GRR}^2+\mathrm{PV}^2}$\\\hline
\end{tabularx}

\stitre{2. Le tableau d'analyse de variance}
Moyennes : $\bar x_{i\cdot\cdot}$ de la pièce $i$ (sur $kr$ mesures), $\bar x_{\cdot j\cdot}$ de l'opérateur $j$ (sur $nr$), $\bar x_{ij\cdot}$ de la cellule (sur $r$), $\bar x_{\cdot\cdot\cdot}$ générale.\par\vspace{3pt}
\renewcommand{\arraystretch}{1.28}
\begin{tabularx}{\linewidth}{|>{\bfseries\raggedright\arraybackslash}p{24mm}|c|>{\raggedright\arraybackslash}X|c|}\hline
\rowcolor{uimm}\color{white}\textbf{Source} & \color{white}\textbf{ddl} & \color{white}\textbf{Somme des carrés $\SC{}$} & \color{white}\textbf{Espérance de $\CM{}=\SC{}/\mathrm{ddl}$}\\\hline
Pièces & $n-1$ & $kr\sum_i(\bar x_{i\cdot\cdot}-\bar x_{\cdot\cdot\cdot})^2$ & $\sigma^2+r\sigma_{PO}^2+kr\sigma_P^2$\\\hline
Opérateurs & $k-1$ & $nr\sum_j(\bar x_{\cdot j\cdot}-\bar x_{\cdot\cdot\cdot})^2$ & $\sigma^2+r\sigma_{PO}^2+nr\sigma_O^2$\\\hline
Interaction & $(n-1)(k-1)$ & $r\sum_{i,j}(\bar x_{ij\cdot}-\bar x_{i\cdot\cdot}-\bar x_{\cdot j\cdot}+\bar x_{\cdot\cdot\cdot})^2$ & $\sigma^2+r\sigma_{PO}^2$\\\hline
Répétabilité & $nk(r-1)$ & $\sum_{i,j,l}(x_{ijl}-\bar x_{ij\cdot})^2$ & $\sigma^2$\\\hline
Total & $nkr-1$ & $\sum_{i,j,l}(x_{ijl}-\bar x_{\cdot\cdot\cdot})^2$ & \\\hline
\end{tabularx}
\par\smallskip
La dernière colonne donne les estimateurs : on résout ces équations pour $\sigma^2$, $\sigma_O^2$, $\sigma_P^2$.

\stitre{3. Du tableau au \%GRR}
\begin{enumerate}[leftmargin=14pt,itemsep=2pt,topsep=2pt]
\item \textbf{Interaction.} $F=\CM{PO}/\CM{E}$, avec $(n-1)(k-1)$ et $nk(r-1)$ ddl. Si $p>0{,}05$ (non significative), on la regroupe avec la répétabilité : $\CM{E}'=\dfrac{\SC{PO}+\SC{E}}{\mathrm{ddl}_{PO}+\mathrm{ddl}_{E}}$. C'est ce que fait Ellistat (ligne «\,interaction 0\,»).
\item \textbf{Écarts types.} $\mathrm{EV}=\sqrt{\CM{E}'}$ \quad $\mathrm{AV}=\sqrt{\dfrac{\CM{O}-\CM{E}'}{nr}}$ \quad $\mathrm{PV}=\sqrt{\dfrac{\CM{P}-\CM{E}'}{kr}}$. Sous la racine, un nombre négatif est ramené à 0.
\item \textbf{Combiner.} $\mathrm{GRR}=\sqrt{\mathrm{EV}^2+\mathrm{AV}^2}$ \quad $\mathrm{TV}=\sqrt{\mathrm{GRR}^2+\mathrm{PV}^2}$.
\item \textbf{Comparer à la tolérance $T$.} $\%\mathrm{GRR}=\dfrac{6\,\mathrm{GRR}}{T}$ (idem \%EV, \%AV) \quad $C_{pc}=\dfrac{T}{6\,\mathrm{GRR}}=\dfrac{100}{\%\mathrm{GRR}}$ \quad $ndc=\sqrt2\,\dfrac{\mathrm{PV}}{\mathrm{GRR}}$ \quad $\dfrac{\mathrm{GRR}}{\mathrm{TV}}$ (part de la variation totale).
\item \textbf{Sans regroupement} (interaction significative) : $\mathrm{EV}=\sqrt{\CM{E}}$, $\mathrm{AV}=\sqrt{(\CM{O}-\CM{PO})/(nr)}$, $\mathrm{INT}=\sqrt{(\CM{PO}-\CM{E})/r}$, $\mathrm{GRR}=\sqrt{\mathrm{EV}^2+\mathrm{AV}^2+\mathrm{INT}^2}$.
\end{enumerate}
