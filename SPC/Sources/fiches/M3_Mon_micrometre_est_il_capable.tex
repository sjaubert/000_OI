\entete{M3}{Mon micromètre est-il capable ?}{Peut-on se fier à ce micromètre pour contrôler la butée ?}{Le moyen de mesure}{40 min}{en binôme}{2}
\begin{tcolorbox}[boite]\textbf{Le but.} Mener une étude de justesse dans Ellistat : 25 mesures d'une cale, puis Cg et Cgk.\par\textbf{Il vous faut.} Un micromètre 0-25 mm (noter son numéro) · Une cale étalon de 10 mm · Ellistat, un projet par stagiaire\end{tcolorbox}
\begin{tcolorbox}[pred]Votre micromètre sera-t-il déclaré capable pour une tolérance de 0,15 mm ? Et pour 0,02 mm ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item Mesurer la cale \textbf{25 fois}, au cliquet, en la retirant à chaque fois. Noter les valeurs dans la grille ci-dessous.
\item \tagE\ Nouvelle grille « Justesse ». Saisir les 25 mesures dans une colonne.
\item \tagE\ Menu \menu{Gage R\&R} > \menu{Justesse}, créer l'étude à partir de la grille.
\item \tagE\ Saisir : valeur vraie $X_0$ = valeur de la cale ; tolérance mini 11,85 et maxi 12,00 (celles de la butée) ; résolution 0,001.
\item \tagE\ Lire les résultats : chaque indicateur est vert, orange ou rouge.
\end{enumerate}

\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item Écart de justesse = \champ{16mm} \quad Cg = \champ{12mm} \quad Cgk = \champ{12mm} \quad Tolérance mini = \champ{16mm}\par\vspace{5pt}
\item Tous les voyants sont-ils verts ? Le micromètre est-il capable pour la butée ?\reponse[1]
\item Cg regarde si les mesures sont groupées, Cgk ajoute le décalage. Lequel de vos deux indices est le plus petit ? Pourquoi ?\reponse[2]
\end{enumerate}
\begin{tcolorbox}[retenir]Capable si Cg et Cgk dépassent 1,33. Cg bon et Cgk faible : l'instrument est fidèle mais déréglé.\end{tcolorbox}
