\entete{M4}{Pied à coulisse contre micromètre}{Deux instruments, une cale : lequel croire ?}{Le moyen de mesure}{30 min}{en binôme}{2}
\begin{tcolorbox}[boite]\textbf{Le but.} Comparer deux instruments sur la même cale : résolution, dispersion, décalage.\par\textbf{Il vous faut.} Un micromètre 0-25 mm · Un pied à coulisse au 1/50 · Une cale étalon de 10 mm\end{tcolorbox}
\begin{tcolorbox}[pred]Lequel des deux instruments donnera les mesures les plus groupées ? De combien, à votre avis ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item Mesurer la cale 15 fois au micromètre (ou reprendre les mesures de M3), puis 15 fois au pied à coulisse. Retirer la cale à chaque fois.
\item \tagE\ \menu{Gage R\&R}, onglet \menu{Justesse}, « Ajouter une justesse » : une étude par instrument. Nombre de mesures 15 ; Valeur vraie = valeur de la cale ; LSI = cale $-$ 0,075 et LSS = cale + 0,075 ; k 3. Saisir toutes les décimales lues : 10,00 et 10,02 au pied à coulisse, 10,000 au micromètre.
\item \tagW\ Coller les deux séries dans la page « Comparer deux séries de mesures ».
\end{enumerate}
