\entete{C1}{La carte sonne-t-elle ?}{Normal ou alarme ? Quelle règle ? Que faire ?}{Cartes aux mesures et capabilité}{30 min}{en binôme}{1}
\begin{tcolorbox}[boite]\textbf{Le but.} S'entraîner à lire une carte de contrôle et à choisir la bonne réaction.\par\textbf{Il vous faut.} La page « La carte sonne-t-elle ? »\end{tcolorbox}
\begin{tcolorbox}[pred]Tous les points sont dans les limites. Peut-il quand même y avoir une alarme ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item \tagW\ Ouvrir la page. Regarder les quatre démonstrations : procédé normal, 1 point hors limite, 7 points du même côté, 6 points qui montent.
\item \tagW\ Série 1 : 10 cartes, aide-mémoire ouvert. En binôme, dire la règle et la réaction à voix haute \textbf{avant} de cliquer.
\item \tagW\ Série 2 : 10 cartes, aide-mémoire fermé, chacun son tour.
\item \tagW\ Essayer les trois « pièges » du menu.
\end{enumerate}

\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item Complétez : 7 points du même côté de la ligne centrale $\rightarrow$ on \champ{60mm}.\par\vspace{5pt}
\item Les limites de contrôle sont-elles les tolérances du plan ? D'où viennent-elles ?\reponse[2]
\item Pourquoi ne faut-il rien régler quand aucune règle ne sonne ?\reponse[2]
\end{enumerate}
\begin{tcolorbox}[retenir]Les limites de contrôle disent ce que la machine fait d'habitude. Les tolérances disent ce que le client accepte. Ce ne sont pas les mêmes lignes.\end{tcolorbox}
