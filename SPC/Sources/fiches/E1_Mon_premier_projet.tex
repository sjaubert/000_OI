\entete{E1}{Mon premier projet}{Où range-t-on son travail dans Ellistat ?}{Prise en main d'Ellistat}{10 min}{seul}{1}
\begin{tcolorbox}[boite]\textbf{Le but.} Ouvrir Data Analysis, créer un projet à son nom et se repérer dans le menu.\par\textbf{Il vous faut.} Un poste connecté et un compte Ellistat par stagiaire\end{tcolorbox}
\begin{tcolorbox}[pred]Dans un tableur, on travaille dans un classeur. À votre avis, comment s'appelle l'équivalent dans Ellistat ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item \tagE\ Sur la page d'accueil, cliquer sur la tuile \menu{Data Analysis}. La \menu{Liste de mes projets} s'ouvre.
\item \tagE\ Cliquer sur \menu{Créer un nouveau projet}. Donner un nom (par exemple « Essais Dupont ») et un commentaire, puis valider. La grille du projet s'ouvre : elle s'appelle « Default ».
\item \tagE\ Dans le bandeau de gauche, repérer les deux blocs : \menu{Gérer} (projets et grilles) et \menu{Analyse} (les calculs).
\item \tagE\ Cliquer sur trois menus du bloc \menu{Analyse}, au choix. Revenir à la grille par \menu{Grilles du projet}.
\item \tagE\ Revenir à \menu{Liste de mes projets}. Retrouver son projet dans « Créé récemment ».
\end{enumerate}
\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item Complétez : un \champ{22mm} contient une ou plusieurs \champ{22mm}, et chaque analyse lit la \champ{22mm} active.\par\vspace{5pt}
\item Citez trois menus du bloc \menu{Analyse}. Lequel servira à valider un instrument de mesure ?\reponse[2]
\item Une page d'analyse affiche « sélection d'un projet ». Que s'est-il passé ? Que faire ?\reponse[1]
\end{enumerate}
\begin{tcolorbox}[retenir]Un projet contient des grilles. On va toujours de \menu{Gérer} (les données) vers \menu{Analyse} (les calculs).\end{tcolorbox}
