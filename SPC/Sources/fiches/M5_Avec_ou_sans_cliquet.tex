\entete{M5}{Avec ou sans cliquet}{La façon de serrer change-t-elle la mesure ?}{Le moyen de mesure}{30 min}{en binôme}{2}
\begin{tcolorbox}[boite]\textbf{Le but.} Montrer que la méthode de mesure fait partie du moyen de mesure.\par\textbf{Il vous faut.} Un micromètre 0-25 mm à cliquet · Une cale étalon de 10 mm\end{tcolorbox}
\begin{tcolorbox}[pred]Si l'on serre le micromètre à la main, sans le cliquet, les valeurs seront-elles plus grandes, plus petites, plus dispersées ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item Même opérateur, même cale, même micromètre. Série A : 15 mesures \textbf{au cliquet} (toujours le même nombre de crans).
\item Série B : 15 mesures \textbf{sans le cliquet}, en serrant le tambour à la main « comme d'habitude ».
\item \tagW\ Coller les deux séries dans la page « Comparer deux séries de mesures », avec la valeur de la cale et $T = 0{,}15$.
\item \tagE\ \menu{Gage R\&R}, onglet \menu{Justesse} : une étude par série (Nombre de mesures 15 ; Valeur vraie = valeur de la cale ; LSI = cale $-$ 0,075 et LSS = cale + 0,075 ; k 3).
\end{enumerate}

\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item Écart entre les deux moyennes = \champ{18mm} mm. La page le juge-t-elle réel ?\reponse[1]
\item Laquelle des deux séries est la plus dispersée ? Comparez les deux Cg.\reponse[1]
\item Rédigez en une phrase la consigne de mesure à afficher au poste.\reponse[2]
\end{enumerate}
\begin{tcolorbox}[retenir]Le moyen de mesure, c'est l'instrument \textbf{et} la façon de s'en servir. Une consigne écrite vaut souvent mieux qu'un nouvel instrument.\end{tcolorbox}
