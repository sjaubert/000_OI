\entete{A1}{Défectueuse ou défauts ?}{Que compte-t-on, et quelle carte choisir ?}{Cartes aux attributs}{15 min}{en binôme}{1}
\begin{tcolorbox}[boite]\textbf{Le but.} Distinguer une pièce défectueuse d'un défaut, et choisir la carte p, np, c ou u.\par\textbf{Il vous faut.} La page « Compter les défauts »\end{tcolorbox}
\begin{tcolorbox}[pred]Une pièce porte trois rayures. Compte-t-elle pour un ou pour trois ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item Pour chaque situation du tableau, décider : compte-t-on des pièces défectueuses ou des défauts ? La taille contrôlée est-elle toujours la même ?
\item En déduire la carte : \textbf{p} ou \textbf{np} (pièces défectueuses), \textbf{c} ou \textbf{u} (défauts).
\item \tagW\ Vérifier avec la page « Compter les défauts », partie « Quelle carte pour quelle situation ? ».
\end{enumerate}

\titre{J'observe, je décide}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{Q\arabic*.}},leftmargin=22pt]
\item La pièce aux trois rayures compte pour \champ{8mm} sur une carte p et pour \champ{8mm} sur une carte c.\par\vspace{5pt}
\item Quand peut-on utiliser la carte np ? Et la carte c ?\reponse[2]
\end{enumerate}
\begin{tcolorbox}[retenir]Pièces défectueuses : cartes p et np. Défauts : cartes c et u. Taille toujours la même : np ou c. Taille variable : p ou u.\end{tcolorbox}
