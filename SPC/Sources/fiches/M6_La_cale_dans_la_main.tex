\entete{M6}{La cale dans la main}{La chaleur des doigts se voit-elle au micromètre ?}{Le moyen de mesure}{25 min}{en binôme}{2}
\begin{tcolorbox}[boite]\textbf{Le but.} Voir l'effet de la température sur une mesure au millième.\par\textbf{Il vous faut.} Un micromètre 0-25 mm · Une cale étalon de 25 mm (ou la plus longue disponible) · Un chronomètre\end{tcolorbox}
\begin{tcolorbox}[pred]Une cale d'acier de 25 mm tenue 3 minutes dans la main fermée s'allonge-t-elle ? De combien : 0,1 µm, 3 µm, 30 µm ?\reponse\end{tcolorbox}
\titre{À faire}
\begin{enumerate}[label=\textcolor{cyan}{\textbf{\arabic*.}},leftmargin=15pt,itemsep=1.5pt]
\item Mesurer 3 fois la cale à la température de la salle. Noter la moyenne : c'est la référence.
\item Garder la cale 3 minutes dans la main fermée.
\item La poser et la mesurer aussitôt, puis toutes les 30 secondes pendant 5 minutes, sans la reprendre en main.
\item Tracer les points sur le graphique : temps en abscisse, mesure en ordonnée.
\end{enumerate}
